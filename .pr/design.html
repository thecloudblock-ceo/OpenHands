+import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
+import { getRegisteredBackends } from "#/api/backend-registry/active-store";
+import type { Backend } from "#/api/backend-registry/types";
+import SharedAutomationSectionsService, {

+  sharedAutomationAssignmentId,
+} from "#/api/shared-automation-sections-service";
+import type { AutomationSectionCommand } from "#/utils/automation-sections";
+import type { GlobalAutomationDashboard } from "./use-global-automation-dashboard";

+const SHARED_SECTIONS_QUERY_KEY = ["shared-automation-sections"] as const;
+
+function assertCurrent(backend: Backend) {
+  const current = getRegisteredBackends().find(
+    (node) => node.id === backend.id,
+  );
+  if (
+    current?.host !== backend.host ||
+    current.apiKey !== backend.apiKey ||
+    current.connectionRevision !== backend.connectionRevision
+  )
+    throw new Error("Backend connection changed; refresh before trying again");
+}
+
+export function useSharedAutomationSections(global: GlobalAutomationDashboard) {
+  const { nodes, origins } = global;
+  const owner = sharedAutomationSectionsOwner(nodes);
+  const client = useQueryClient();
+  // No credentials in keys; revisions invalidate reads when credentials change.
+  const queryKey = [
+    ...SHARED_SECTIONS_QUERY_KEY,
+    owner?.id,
+    nodes.map((node) => [node.id, node.connectionRevision ?? 0]),
+  ];
+  // eslint-disable-next-line @tanstack/query/exhaustive-deps
+  const query = useQuery({
+    queryKey,
+    queryFn: () => SharedAutomationSectionsService.read(owner!, nodes),
+    enabled: !!owner,
+    retry: false,
+    staleTime: 0,
+    refetchOnWindowFocus: true,
+    refetchInterval: 30_000,
+    meta: { disableToast: true },
+  });
+  const mutation = useMutation({
+    mutationFn: (command: AutomationSectionCommand) => {
+      if (!owner) throw new Error("No section storage backend");
+      assertCurrent(owner);
+      let edit = command;
+      if (command.type === "assign") {
+        const origin = origins.get(command.automationId);
+        if (!origin) throw new Error("Automation is no longer available");
+        assertCurrent(origin.backend);
+        edit = {
+          ...command,
+          automationId: sharedAutomationAssignmentId(origin.backend, origin.id),
+        };
+      }
+      return SharedAutomationSectionsService.update(owner, nodes, edit);
+    },
+    scope: { id: `${SHARED_SECTIONS_QUERY_KEY[0]}:${owner?.id}` },
+    onMutate: async () => {
+      await client.cancelQueries({ queryKey });
+      return { queryKey };
+    },
+    onSuccess: (state, _command, context) =>
+      client.setQueryData(context.queryKey, state),
+    onSettled: () =>
+      client.invalidateQueries({ queryKey: SHARED_SECTIONS_QUERY_KEY }),
+    meta: { disableToast: true },
+  });
+  return {
+    scopeKey: `${owner?.id}:${owner?.connectionRevision ?? 0}`,
+    data: query.data && {
+      ...query.data,
+      assignments: Object.fromEntries(
+        [...origins].map(([key, origin]) => [
+          key,
+          query.data.assignments[
+            sharedAutomationAssignmentId(origin.backend, origin.id)
+          ] ?? null,
+        ]),
+      ),
+    },
+    isError: query.isError,
+    isSuccess: query.isSuccess,
+    refetch: query.refetch,
+    supported: !!owner,
+    mutation,
+  };

+/** Only this service writes the shared namespace, on the hosting Canvas backend. */
+export default class SharedAutomationSectionsService {
+  private static async load(owner: Backend, nodes: Backend[]) {
+    const settings = await automationSectionsClient(owner).getSettings();
+    const saved = readSharedState(settings);
+    const state = structuredClone(saved);
+    const isFirstImport = !Object.hasOwn(
+      settings.misc_settings ?? {},
+      SHARED_AUTOMATION_SECTIONS_KEY,
+    );
+    let unassignedCollapsed: boolean | undefined;
+    // Successful imports are persisted with the next explicit edit. Failed reads
+    // stay unimported and are retried; never turn a failed read into an empty import.
+    const pending = [
+      owner,
+      ...nodes
+        .filter((node) => node.id !== owner.id)
+        .sort((a, b) =>
+          automationSectionHost(a).localeCompare(automationSectionHost(b)),
+        ),
+    ].filter((node) => !state.importedHosts[automationSectionHost(node)]);
+    const legacy = await Promise.allSettled(
+      pending.map((node) =>
+        node.id === owner.id
+          ? Promise.resolve().then(() => readAutomationSectionsState(settings))
+          : AutomationSectionsService.read(node),
+      ),
+    );
+    const importedSections: AutomationSectionsState["sections"] = {};
+    const importedAssignments: AutomationSectionsState["assignments"] = {};
+    const importedHosts: Record<string, true> = {};
+    legacy.forEach((result, index) => {
+      if (result.status !== "fulfilled") return;
+      const node = pending[index];
+      const old = result.value;
+      if (isFirstImport && node.id === owner.id) {
+        unassignedCollapsed = old.unassignedCollapsed;
+        state.unassignedCollapsed = old.unassignedCollapsed;
+      }
+      const ids = new Map<string, string>();

+        ids.set(id, sharedId);
+        if (!Object.hasOwn(state.sections, sharedId)) {
+          state.sections[sharedId] = section;
+          importedSections[sharedId] = section;
+        }
+      }
+      for (const id of Object.keys(old.assignments)) {
+        const key = sharedAutomationAssignmentId(node, id);
+        if (Object.hasOwn(state.assignments, key)) continue;

+        const section = oldSection ? (ids.get(oldSection) ?? null) : null;
+        state.assignments[key] = section;
+        importedAssignments[key] = section;
+      }
+      importedHosts[automationSectionHost(node)] = true;
+      state.importedHosts[automationSectionHost(node)] = true;
+    });
+    return {
+      state,
+      patch: {
+        version: 1,
+        sections: importedSections,
+        assignments: importedAssignments,
+        importedHosts,

+      },
+    };
+  }

Evidence & limits
